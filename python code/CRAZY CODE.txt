# --- 3. Mathematical Conversions ---
def calculate_mass_from_einstein_radius(theta_E_arcsec):
    """
    Rearranges the Einstein Radius formula to solve for Mass (M).
    M = (theta_E^2 * c^2 * D_L * D_S) / (4 * G * D_LS)
    """
    # Convert arcseconds to radians
    theta_E_rad = theta_E_arcsec * (np.pi / (180 * 3600))
    
    # Calculate mass in kg
    mass_kg = (theta_E_rad**2 * c**2 * D_L * D_S) / (4 * G * D_LS)
    return mass_kg

# --- 4. RK4 Differential Equations ---
def get_derivatives(state, M):
    x, y, vx, vy = state
    r = np.sqrt(x**2 + y**2)
    
    # General Relativity correction for a point mass
    a_mag = (2 * G * M) / (r**2)
    ax = -a_mag * (x / r)
    ay = -a_mag * (y / r)
    
    return np.array([vx, vy, ax, ay])

def rk4_step(state, dt, M):
    k1 = get_derivatives(state, M)
    k2 = get_derivatives(state + 0.5 * dt * k1, M)
    k3 = get_derivatives(state + 0.5 * dt * k2, M)
    k4 = get_derivatives(state + dt * k3, M)
    
    return state + (dt / 6.0) * (k1 + 2*k2 + 2*k3 + k4)

def simulate_deflection(mass_kg, impact_parameter_m):
    """Runs the RK4 simulation at the physical distance of the Einstein radius."""
    x_start = -10.0 * impact_parameter_m
    state = np.array([x_start, impact_parameter_m, c, 0.0]) # [x, y, vx, vy]
    
    # Step size dynamically scaled based on the vast galactic distances
    dt = (impact_parameter_m / c) * 0.05 
    
    # Integrate until the photon travels far past the galaxy
    while state[0] < 10.0 * impact_parameter_m:
        state = rk4_step(state, dt, mass_kg)

    vx_final, vy_final = state[2], state[3]
    deflection_radians = np.abs(np.arctan2(vy_final, vx_final))
    return deflection_radians * (180 * 3600 / np.pi) # Return in arcsec

# --- 5. Run the Pipeline on all 59 Systems ---
results = []
print(f"Processing all {len(slacs_data)} SLACS Lenses via RK4 Integration. Please wait...\n")

for row in slacs_data:
    system = row["System"]
    b_sie = row["b_SIE"]
    
    # 1. Reverse-engineer the mass from the table's Einstein Radius
    estimated_mass_kg = calculate_mass_from_einstein_radius(b_sie)
    mass_in_solar = estimated_mass_kg / M_sun
    
    # 2. Calculate physical impact parameter in meters
    b_sie_rad = b_sie * (np.pi / (180 * 3600))
    impact_parameter_m = b_sie_rad * D_L
    
    # 3. Run RK4 Simulation
    rk4_deflection = simulate_deflection(estimated_mass_kg, impact_parameter_m)
    
    results.append({
        "System Name": system,
        "Measured b_SIE (arcsec)": b_sie,
        "Est. Mass (Trillion M_sun)": mass_in_solar / 1e12,
        "RK4 Deflection (arcsec)": rk4_deflection
    })
